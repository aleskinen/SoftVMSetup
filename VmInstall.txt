cat > /etc/nftables.conf
#!/usr/sbin/nft -f

flush ruleset

table inet filter {
    # Dynamic set to hold banned IPv4 addresses for 10 minutes
    set blocklist_v4 {
        type ipv4_addr
        flags dynamic, timeout
        timeout 10m
    }

    # Dynamic set to hold banned IPv6 addresses for 10 minutes
    set blocklist_v6 {
        type ipv6_addr
        flags dynamic, timeout
        timeout 10m
    }

    chain input {
        type filter hook input priority 0; policy drop;

        # 1. Allow loopback and established/related traffic
        iifname "lo" accept
        ct state established,related accept

        # 2. Drop invalid packets early (stops malformed packet floods)
        ct state invalid drop

        # 3. Drop traffic from already-banned IPs
        ip saddr @blocklist_v4 drop
        ip6 saddr @blocklist_v6 drop

        # 4. Rate-limit ICMP (Ping)
        ip protocol icmp icmp type echo-request limit rate 2/second burst 5 packets accept
        ip6 nexthdr icmpv6 icmpv6 type echo-request limit rate 2/second burst 5 packets accept

        # 5. Protect SSH (Max 3 new connections per minute per IP)
        tcp dport 22 ct state new meter ssh_limit_v4 { ip saddr limit rate over 3/minute burst 3 packets } add @blocklist_v4 { ip saddr timeout 1h } drop
        tcp dport 22 ct state new meter ssh_limit_v6 { ip6 saddr limit rate over 3/minute burst 3 packets } add @blocklist_v6 { ip6 saddr timeout 1h } drop
        tcp dport 22 accept

        # 6. Protect HTTP/HTTPS against SYN/Connection floods
        # Ban IPs opening more than 25 new connections/sec (burst 50)
        tcp dport { 80, 443 } ct state new meter web_rate_v4 { ip saddr limit rate over 25/second burst 50 packets } add @blocklist_v4 { ip saddr } drop
        tcp dport { 80, 443 } ct state new meter web_rate_v6 { ip6 saddr limit rate over 25/second burst 50 packets } add @blocklist_v6 { ip6 saddr } drop

        # Limit max concurrent active connections per IP to 50
        tcp dport { 80, 443 } ct state new meter web_conn_v4 { ip saddr ct count over 50 } drop
        tcp dport { 80, 443 } ct state new meter web_conn_v6 { ip6 saddr ct count over 50 } drop

        # Allow legitimate web traffic
        tcp dport { 80, 443 } accept
    }

    chain forward {
        type filter hook forward priority 0; policy drop;
    }

    chain output {
        type filter hook output priority 0; policy accept;
    }
}
